\section{Understanding LQR}
Consider the following LQR formulation
\begin{align*}
    \min_u\quad &\sum_{k=0}^{N-1}x_{k}^{\top}Qx_{k} + u_{k}^{\top}Ru_k + x_{N}^{\top}Px_N,\\
    \text{s.t.}\quad &x_{k+1} = Ax_k + Bu_k, \forall k \in [N-1].
\end{align*}
\begin{enumerate}
    \item Explain how the controller will behave when 1) $Q = 0, P = 0$. 2) $R= 0$.\vspace{3\baselineskip}
    \item Explain the difference between finite horizon $LQR$ and infinite horizon $LQR$ ($N = \infty$).Why is the infinite horizon controller time invariant?\vspace{3\baselineskip}
\end{enumerate}
\section{Dynamic programming}
Consider the following optimal control problem
\begin{align*}
    V^*(x_0) = \min_u \quad &\gamma x_N^2 + \sum_{i = 0}^{2}u_i^2,\\
    \text{s.t.}\quad&x_{k+1} = x_k+ u_k,
\end{align*}

where only the cost of the final state is penalized with $\gamma \geq 0$. Your goal is to solve this problem with dynamic programming.
\begin{enumerate}
    \item Assume that $V_{k+1}(x_{k+1}) = hx_{k+1}^2$ and derive $V_k(x_k)$.\vspace{3\baselineskip}
    \item What is $V_N(x_N)$?\vspace{3\baselineskip}
    \item Compute the optimal value function $V^*(x_0)$ and the control law $\kappa^*(x_0)$ for a finite horizon $N$.\vspace{3\baselineskip}
    \item Compute the closed-loop pole of your system. Will your system converge to zero for all finite $N$ and $\gamma$?\vspace{3\baselineskip}
    \item What is your control law when $N\rightarrow \infty$. Will you system converge to zero for infinite $N$ and finite $\gamma$?
\end{enumerate}
